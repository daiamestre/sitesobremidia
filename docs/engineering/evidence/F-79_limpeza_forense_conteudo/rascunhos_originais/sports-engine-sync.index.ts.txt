const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

const admin = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

function json(status: number, body: object): Response {
  return new Response(JSON.stringify(body), {
    status,
    headers: { "Content-Type": "application/json", ...corsHeaders },
  });
}

function normalizeStatus(raw: string): { status: string; display: string } {
  const s = (raw || "").toUpperCase().trim();
  switch (s) {
    case "FT":
    case "AET":
    case "PEN":
    case "FINISHED":
      return { status: "FINISHED", display: "Encerrado" };
    case "1H":
    case "2H":
    case "ET":
    case "LIVE":
      return { status: "LIVE", display: "Ao Vivo" };
    case "HT":
    case "HALFTIME":
      return { status: "HALFTIME", display: "Intervalo" };
    case "PST":
    case "POSTPONED":
      return { status: "POSTPONED", display: "Adiado" };
    case "CANC":
    case "CANCELLED":
      return { status: "CANCELLED", display: "Cancelado" };
    case "SUSP":
    case "SUSPENDED":
      return { status: "SUSPENDED", display: "Suspenso" };
    case "NS":
    case "TBD":
    case "SCHEDULED":
    default:
      return { status: "SCHEDULED", display: "A Definir" };
  }
}

serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    let tenantId: string | null = null;
    const authHeader = req.headers.get("Authorization");

    if (authHeader && authHeader.startsWith("Bearer ")) {
      const token = authHeader.replace("Bearer ", "");
      const { data: { user } } = await admin.auth.getUser(token);
      if (user) {
        const { data: userData } = await admin
          .from("usuarios")
          .select("empresa_operadora_id")
          .eq("id", user.id)
          .single();
        if (userData?.empresa_operadora_id) {
          tenantId = userData.empresa_operadora_id;
        }
      }
    }

    let body: { competition_slug?: string; tenant_id?: string } = {};
    if (req.method === "POST") {
      try { body = await req.json(); } catch { /* ignore */ }
    }

    if (body.tenant_id) tenantId = body.tenant_id;

    if (!tenantId) {
      const { data: tenant } = await admin
        .from("empresa_operadora")
        .select("id")
        .limit(1)
        .single();
      tenantId = tenant?.id || null;
    }

    if (!tenantId) {
      return json(400, { error: "tenant_nao_encontrado" });
    }

    let query = admin
      .from("content_sports_competitions")
      .select("id, empresa_operadora_id, external_id, nome, slug, temporada, source")
      .eq("empresa_operadora_id", tenantId)
      .eq("ativo", true);

    if (body.competition_slug) {
      query = query.eq("slug", body.competition_slug);
    }

    const { data: competitions, error: compErr } = await query;
    if (compErr) throw compErr;

    let totalFixtures = 0;
    const results = [];

    for (const comp of competitions || []) {
      try {
        let fixturesToUpsert: any[] = [];

        // 1. If API_FOOTBALL_KEY is set, try fetching live data from api-football
        if (API_FOOTBALL_KEY && comp.external_id) {
          const season = comp.temporada || new Date().getFullYear().toString();
          const apiUrl = `https://v3.football.api-sports.io/fixtures?league=${comp.external_id}&season=${season}&last=5`;
          const res = await fetch(apiUrl, {
            headers: {
              "x-rapidapi-key": API_FOOTBALL_KEY,
              "x-rapidapi-host": "v3.football.api-sports.io",
            },
            signal: AbortSignal.timeout(10000),
          });

          if (res.ok) {
            const apiData = await res.json();
            if (apiData?.response && Array.isArray(apiData.response) && apiData.response.length > 0) {
              fixturesToUpsert = apiData.response.map((f: any) => {
                const norm = normalizeStatus(f.fixture?.status?.short);
                const kickoff = f.fixture?.date ? new Date(f.fixture.date).toISOString() : new Date().toISOString();
                const matchDate = kickoff.slice(0, 10);

                return {
                  empresa_operadora_id: comp.empresa_operadora_id,
                  competition_id: comp.id,
                  external_fixture_id: String(f.fixture?.id),
                  temporada: season,
                  rodada: f.league?.round || null,
                  match_date: matchDate,
                  kickoff_utc: kickoff,
                  home_team_id: String(f.teams?.home?.id || ""),
                  home_team_name: f.teams?.home?.name || "",
                  home_team_logo: f.teams?.home?.logo || null,
                  away_team_id: String(f.teams?.away?.id || ""),
                  away_team_name: f.teams?.away?.name || "",
                  away_team_logo: f.teams?.away?.logo || null,
                  home_score: f.goals?.home ?? null,
                  away_score: f.goals?.away ?? null,
                  status: norm.status,
                  status_display: norm.display,
                  venue: f.fixture?.venue?.name || null,
                  source: "api-football",
                  raw_status: f.fixture?.status?.short || null,
                };
              });
            }
          }
        }

        // 2. Fallback to baseline structured fixtures if external API didn't return fixtures
        if (fixturesToUpsert.length === 0 && BASELINE_FIXTURES[comp.slug]) {
          const list = BASELINE_FIXTURES[comp.slug];
          fixturesToUpsert = list.map((item) => {
            const norm = normalizeStatus(item.status);
            const kickoff = item.kickoff_utc;
            const matchDate = kickoff.slice(0, 10);

            return {
              empresa_operadora_id: comp.empresa_operadora_id,
              competition_id: comp.id,
              external_fixture_id: item.external_id,
              temporada: comp.temporada || new Date().getFullYear().toString(),
              rodada: item.rodada,
              match_date: matchDate,
              kickoff_utc: kickoff,
              home_team_name: item.home_team,
              away_team_name: item.away_team,
              home_score: item.home_score ?? null,
              away_score: item.away_score ?? null,
              status: norm.status,
              status_display: norm.display,
              source: "baseline",
            };
          });
        }

        let savedCount = 0;
        for (const fix of fixturesToUpsert) {
          const { error: fErr } = await admin
            .from("content_sports_fixtures")
            .upsert(fix, {
              onConflict: "empresa_operadora_id,competition_id,external_fixture_id",
            });
          if (!fErr) savedCount++;
        }

        totalFixtures += savedCount;

        await admin
          .from("content_sports_competitions")
          .update({
            last_sync_at: new Date().toISOString(),
            last_sync_error: null,
            fixtures_count: savedCount,
            updated_at: new Date().toISOString(),
          })
          .eq("id", comp.id);

        results.push({
          competition: comp.nome,
          slug: comp.slug,
          fixtures_synced: savedCount,
          status: "SUCCESS",
        });
      } catch (err: any) {
        await admin
          .from("content_sports_competitions")
          .update({
            last_sync_at: new Date().toISOString(),
            last_sync_error: err?.message || String(err),
            updated_at: new Date().toISOString(),
          })
          .eq("id", comp.id);

        results.push({
          competition: comp.nome,
          slug: comp.slug,
          error: err?.message || String(err),
          status: "ERROR",
        });
      }
    }

    return json(200, {
      success: true,
      tenant_id: tenantId,
      competitions_processed: results.length,
      fixtures_synced: totalFixtures,
      details: results,
    });
  } catch (error: any) {
    console.error("[SportsEngineSync] Fatal error:", error);
    return json(500, { error: error?.message || "internal_error" });
  }
});
